cmake_minimum_required(VERSION 3.20.0)
list(APPEND BOARD_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
list(APPEND SOC_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
list(APPEND DTS_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
set(RPIDUINO_SKETCH heartbeat CACHE STRING "RPiduino sketch to build")
if(RPIDUINO_SKETCH STREQUAL "uart_debug")
  list(APPEND EXTRA_CONF_FILE ${CMAKE_CURRENT_SOURCE_DIR}/sketches/uart_debug.conf)
  list(APPEND EXTRA_DTC_OVERLAY_FILE ${CMAKE_CURRENT_SOURCE_DIR}/sketches/uart_debug.overlay)
endif()
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(rpiduino_zephyr LANGUAGES C CXX)

target_include_directories(app PRIVATE include)
if(RPIDUINO_SKETCH STREQUAL "gpio_loopback" OR RPIDUINO_SKETCH STREQUAL "uart_debug")
  if(NOT CONFIG_SOC_BCM2837)
    message(FATAL_ERROR "${RPIDUINO_SKETCH} requires the rpi_3 board")
  endif()
elseif(NOT RPIDUINO_SKETCH STREQUAL "heartbeat")
  message(FATAL_ERROR "RPIDUINO_SKETCH must be heartbeat, gpio_loopback, or uart_debug")
endif()
target_sources(app PRIVATE src/main.cpp sketches/${RPIDUINO_SKETCH}.cpp)
if(CONFIG_SOC_BCM2837)
  target_sources(app PRIVATE src/gpio_bcm2837.cpp src/uart_control.cpp)
endif()
if(RPIDUINO_SKETCH STREQUAL "uart_debug")
  target_sources(app PRIVATE drivers/sdhc/sdhc_bcm2835_sdhost.c src/image_update.cpp)
endif()
